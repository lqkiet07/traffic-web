```python
def _avg_kpi_per_cycle(path):
    """Average Queue/Delay/Throughput across 8 junctions per cycle."""
    per = {}
    with open(path, encoding="utf-8-sig") as fh:
        for row in csv.DictReader(fh):
            try:
                c = int(float(row["Cycle"]))
                e = per.setdefault(c, {"q": [], "d": [], "t": []})
                e["q"].append(float(row["Queue_Length"]))
                e["d"].append(float(row["Average_Delay"]))
                e["t"].append(float(row["Throughput_per_Cycle"]))
            except (KeyError, ValueError):
                continue
    cycles = sorted(per)
    avg = lambda k: [round(sum(per[c][k]) / max(len(per[c][k]), 1), 4) for c in cycles]
    return cycles, avg("d"), avg("q"), avg("t")


def export_kpi():
    """Build kpi_summary.json from rep1 KPI_Result CSVs (1:1 with replay).

    Medium_900: proposed <- KPI_Result_CBMP_Area_Medium_900_rep1,
    baseline <- KPI_Result_CBMP_Paper_Fair_Medium_900_rep1 (same batch).
    Low_400/High_1400: keep legacy excel_data (different batch, bannered).
    """
    rep1 = {
        "Medium_900": (
            os.path.join(OUTPUTS_DIR, "KPI_Result_CBMP_Area_Medium_900_rep1.csv"),
            os.path.join(OUTPUTS_DIR, "KPI_Result_CBMP_Paper_Fair_Medium_900_rep1.csv"),
        ),
    }
    summary = {}
    for file_key, label in KPI_SCENARIOS:
        if file_key in rep1 and all(os.path.exists(p) for p in rep1[file_key]):
            p_path, b_path = rep1[file_key]
            cycles, p_d, p_q, p_t = _avg_kpi_per_cycle(p_path)
            _, b_d, b_q, b_t = _avg_kpi_per_cycle(b_path)
            print(f"[kpi] {file_key}: rep1 1:1 ({len(cycles)} cycles)")
        else:
            legacy = os.path.join(OUTPUTS_DIR, "plots", f"excel_data_{file_key}.csv")
            cycles, p_d, p_q, p_t, b_d, b_q, b_t = _legacy_kpi(legacy)
            print(f"[kpi] {file_key}: legacy excel ({len(cycles)} cycles, NOT 1:1)")
        summary[file_key] = {
            "label": label,
            "cycles": cycles,
            "proposed": {"delay": p_d, "queue": p_q, "throughput": p_t},
            "baseline": {"delay": b_d, "queue": b_q, "throughput": b_t},
        }
    os.makedirs(WEB_DATA_DIR, exist_ok=True)
    path = os.path.join(WEB_DATA_DIR, "kpi_summary.json")
    with open(path, "w", encoding="utf-8") as fh:
        json.dump(summary, fh, separators=(",", ":"))
    return path
```

`_legacy_kpi(legacy)` = than cu dong 244-261 tach ra nguyen ven (doc `CBMP_Area_*` + `CBMP_Paper_Fair_*`/`CBMP_Paper_*` fallback), return tuple 7 phan tu nhu cu. Khong doi 1 ky tu logic legacy.

// tests/kpi_rep1.test.mjs
import assert from "node:assert";
import { readFileSync } from "node:fs";

const kpi = JSON.parse(readFileSync("public/data/kpi_summary.json", "utf8"));
const jk = JSON.parse(readFileSync("public/data/junction_kpis.json", "utf8"));

const m = kpi.Medium_900;
assert.strictEqual(m.cycles.length, 61, "Medium_900 rep1 1:1 = 61 cycles");
assert.deepStrictEqual(m.cycles, Array.from({ length: 61 }, (_, i) => i + 1));
for (const k of ["delay", "queue", "throughput"]) {
  assert.strictEqual(m.proposed[k].length, 61, `proposed.${k} 61 pts`);
  assert.strictEqual(m.baseline[k].length, 61, `baseline.${k} 61 pts`);
}

const j1 = Object.values(jk.cao.Medium_900).flatMap((arr) => arr.filter((e) => e.cycle === 1));
const avg = (a) => a.reduce((s, v) => s + v, 0) / a.length;
const q = avg(j1.map((e) => e.queue));
assert(Math.abs(q - m.proposed.queue[0]) < 1e-3, `kpi matches junction rep1 (q=${q})`);

assert.strictEqual(kpi.Low_400.cycles.length, 64, "Low legacy untouched");
assert.strictEqual(kpi.High_1400.cycles.length, 64, "High legacy untouched");

console.log("[kpi_rep1.test] all assertions passed");
